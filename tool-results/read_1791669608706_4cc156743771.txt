   382→/** Vehicle speedometer — SVG arc gauge + hull integrity + tank cannon reload. */
   383→function Speedo({ driving, air }: { driving: NonNullable<HudState['driving']>; air: number }) {
   384→  const max = driving.tank ? 60 : 130;
   385→  const pct = Math.max(0, Math.min(1, driving.speed / max));
   386→  const angle = (-90 + pct * 180) * (Math.PI / 180);
   387→  const needleLen = 44;
   388→  const nx = 75 + needleLen * Math.sin(angle);
   389→  const ny = 74 - needleLen * Math.cos(angle);
   390→  const hpct = Math.max(0, Math.min(1, driving.health / (driving.tank ? 400 : 100)));
   391→  const lowHealth = hpct < 0.3;
   392→  return (
   393→    <div
   394→      className={`mt-1 rounded-lg border px-2 pb-1.5 pt-1 backdrop-blur-[2px] ${
   395→        driving.tank ? 'border-lime-400/40 bg-[#0d1206]/75' : 'border-white/15 bg-black/60'
   396→      }`}
   397→    >
   398→      <svg width="150" height="88" viewBox="0 0 150 88">
   399→        {/* gauge arc */}
   400→        <path d="M 23 74 A 52 52 0 0 1 127 74" fill="none" stroke="rgba(255,255,255,0.13)" strokeWidth="7" strokeLinecap="round" />
   401→        <path
   402→          d="M 23 74 A 52 52 0 0 1 127 74"
   403→          fill="none"
   404→          stroke={driving.tank ? '#a3e635' : '#fbbf24'}
   405→          strokeWidth="7"
   406→          strokeLinecap="round"
   407→          strokeDasharray={`${pct * 163.4} 400`}
   408→        />
   409→        {/* ticks */}
   410→        {[0, 0.25, 0.5, 0.75, 1].map((t, i) => {
   411→          const a = (-90 + t * 180) * (Math.PI / 180);
   412→          const x1 = 75 + 40 * Math.sin(a);
   413→          const y1 = 74 - 40 * Math.cos(a);
   414→          const x2 = 75 + 45 * Math.sin(a);
   415→          const y2 = 74 - 45 * Math.cos(a);
   416→          return <line key={i} x1={x1} y1={y1} x2={x2} y2={y2} stroke="rgba(255,255,255,0.4)" strokeWidth="1.6" />;
   417→        })}
   418→        {/* needle */}
   419→        <line x1="75" y1="74" x2={nx} y2={ny} stroke="#ffffff" strokeWidth="2.4" strokeLinecap="round" />
   420→        <circle cx="75" cy="74" r="4" fill="#ffffff" />
   421→        <text x="75" y="58" textAnchor="middle" fill="#ffffff" fontSize="19" fontWeight="800" className="tabular-nums">
   422→          {Math.round(driving.speed)}
   423→        </text>
   424→        <text x="75" y="86" textAnchor="middle" fill="rgba(255,255,255,0.5)" fontSize="8" letterSpacing="3">
   425→          KM/H
   426→        </text>
   427→      </svg>
   428→      {/* hull integrity */}
   429→      <div className="mt-0.5 h-1.5 overflow-hidden rounded-full bg-white/10">
   430→        <div
   431→          className={`h-full rounded-full transition-[width] duration-300 ${
   432→            lowHealth ? 'bg-gradient-to-r from-red-600 to-orange-400' : 'bg-gradient-to-r from-sky-400/80 to-emerald-400'
   433→          }`}
   434→          style={{ width: `${hpct * 100}%` }}
   435→        />
   436→      </div>
   437→      <div className="mt-1 flex items-center justify-between">
   438→        <span className={`text-[9px] font-black uppercase tracking-[0.14em] ${driving.tank ? 'text-lime-300/90' : 'text-amber-300/90'}`}>
   439→          {driving.name}
   440→        </span>

   472→/** Mobile/touch control layer — analog joystick, drag-look, action buttons.
   473→ *  Rendered above the canvas but below chips/hotbar so HUD taps still work. */
   474→function TouchControls({ game, state }: { game: HyperionGame; state: HudState }) {
   475→  const input = game.input;
   476→  const [knob, setKnob] = useState({ x: 0, y: 0 });
   477→  const [sprintOn, setSprintOn] = useState(false);
   478→  const padRef = useRef<HTMLDivElement | null>(null);
   479→  const joyId = useRef<number | null>(null);
   480→  const lookId = useRef<number | null>(null);
   481→  const lookLast = useRef<{ x: number; y: number } | null>(null);
   482→  const R = 50;
   483→
   484→  const moveJoy = (e: React.PointerEvent) => {
   485→    const el = padRef.current;
   486→    if (!el) return;
   487→    const rect = el.getBoundingClientRect();
   488→    let dx = e.clientX - (rect.left + rect.width / 2);
   489→    let dy = e.clientY - (rect.top + rect.height / 2);
   490→    const d = Math.hypot(dx, dy);
   491→    if (d > R) {
   492→      dx = (dx / d) * R;
   493→      dy = (dy / d) * R;
   494→    }
   495→    setKnob({ x: dx, y: dy });
   496→    input.setVirtualMove(dx / R, -dy / R);
   497→  };
   498→
   499→  const endJoy = () => {
   500→    joyId.current = null;
   501→    setKnob({ x: 0, y: 0 });
   502→    input.setVirtualMove(0, 0);
   503→  };
   504→
   505→  // Look layer handlers (full-screen drag area)
   506→  const onLookDown = (e: React.PointerEvent) => {
   507→    if (lookId.current !== null) return;
   508→    lookId.current = e.pointerId;
   509→    lookLast.current = { x: e.clientX, y: e.clientY };
   510→  };
   511→  const onLookMove = (e: React.PointerEvent) => {
   512→    if (lookId.current !== e.pointerId || !lookLast.current) return;
   513→    input.addLook((e.clientX - lookLast.current.x) * 2.1, (e.clientY - lookLast.current.y) * 2.1);
   514→    lookLast.current = { x: e.clientX, y: e.clientY };
   515→  };
   516→  const onLookUp = (e: React.PointerEvent) => {
   517→    if (lookId.current === e.pointerId) {
   518→      lookId.current = null;
   519→      lookLast.current = null;
   520→    }
   521→  };
   522→
   523→  // Generic press-and-hold button that simulates a keyboard key.
   524→  const keyBtn = (label: string, code: string, cls = 'h-14 w-14 text-lg') => (
   525→    <button
   526→      className={`hyperion-touchbtn pointer-events-auto flex items-center justify-center rounded-full border border-white/25 bg-black/45 font-black text-white/90 backdrop-blur-[2px] ${cls}`}
   527→      onPointerDown={(e) => {
   528→        e.preventDefault();
   529→        input.pressCode(code);
   530→      }}
   531→      onPointerUp={() => input.releaseCode(code)}
   532→      onPointerLeave={() => input.releaseCode(code)}
   533→      onPointerCancel={() => input.releaseCode(code)}
   534→      aria-label={label}
   535→    >
   536→      {label}
   537→    </button>
   538→  );
   539→
   540→  const inTank = !!state.driving?.tank;
   541→
   542→  // DIG (on foot, hold) / FIRE (tank tap / horn while driving)
   543→  const digFireBtn = (
   544→    <button
   545→      className="hyperion-touchbtn pointer-events-auto flex h-16 w-16 items-center justify-center rounded-full border border-white/25 bg-black/45 text-xl font-black text-white/90 backdrop-blur-[2px]"
   546→      onPointerDown={(e) => {
   547→        e.preventDefault();
   548→        if (state.driving) input.tapFire();
   549→        else input.setVirtualDig(true);
   550→      }}
   551→      onPointerUp={() => {
   552→        input.setVirtualDig(false);
   553→      }}
   554→      onPointerLeave={() => {
   555→        input.setVirtualDig(false);
   556→      }}
   557→      onPointerCancel={() => {
   558→        input.setVirtualDig(false);
   559→      }}
   560→      aria-label={inTank ? 'fire cannon' : state.driving ? 'horn' : 'dig'}
   561→    >
   562→      {inTank ? '🔥' : state.driving ? '📢' : '⛏'}
   563→    </button>
   564→  );
   565→
   566→  // USE / PLACE — tap places a block or uses the item; hold = jetpack thrust.
   567→  const useBtn = (
   568→    <button
   569→      className="hyperion-touchbtn pointer-events-auto flex h-14 w-14 items-center justify-center rounded-full border border-white/25 bg-black/45 text-lg font-black text-white/90 backdrop-blur-[2px]"
   570→      onPointerDown={(e) => {
   571→        e.preventDefault();
   572→        input.setVirtualUseHold(true);
   573→        input.tapUse();
   574→      }}
   575→      onPointerUp={() => {
   576→        input.setVirtualUseHold(false);
   577→      }}
   578→      onPointerLeave={() => {
   579→        input.setVirtualUseHold(false);
   580→      }}
   581→      onPointerCancel={() => {
   582→        input.setVirtualUseHold(false);
   583→      }}
   584→      aria-label="use item"
   585→    >
   586→      🧱
   587→    </button>
   588→  );
   589→
   590→  return (
   591→    <div className="absolute inset-0" style={{ touchAction: 'none' }}>
   592→      {/* Look-drag layer (below all buttons / HUD elements in DOM order) */}
   593→      <div
   594→        className="absolute inset-0"
   595→        onPointerDown={onLookDown}
   596→        onPointerMove={onLookMove}
   597→        onPointerUp={onLookUp}
   598→        onPointerCancel={onLookUp}
   599→      />
   600→
   601→      {/* Analog joystick (bottom-left) */}
   602→      <div
   603→        ref={padRef}
   604→        className="hyperion-touchbtn pointer-events-auto absolute bottom-24 left-5 h-[124px] w-[124px] rounded-full border-2 border-white/20 bg-black/35 backdrop-blur-[2px]"
   605→        onPointerDown={(e) => {
   606→          e.preventDefault();
   607→          joyId.current = e.pointerId;
   608→          (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
   609→          moveJoy(e);
   610→        }}
   611→        onPointerMove={(e) => {
   612→          if (joyId.current === e.pointerId) moveJoy(e);
   613→        }}
   614→        onPointerUp={endJoy}
   615→        onPointerCancel={endJoy}
   616→        aria-label="movement joystick"
   617→      >
   618→        <div className="absolute inset-0 flex items-center justify-center">
   619→          <div
   620→            className="h-12 w-12 rounded-full border border-white/40 bg-white/25 shadow-[0_0_12px_rgba(255,255,255,0.25)] transition-transform duration-75"
   621→            style={{ transform: `translate(${knob.x}px, ${knob.y}px)` }}
   622→          />
   623→        </div>
   624→        <div className="absolute -top-5 left-1/2 -translate-x-1/2 text-[9px] font-black uppercase tracking-widest text-white/40">
   625→          move
   626→        </div>
   627→      </div>
   628→
   629→      {/* Action cluster (bottom-right) */}
   630→      <div className="absolute bottom-24 right-5 flex flex-col items-end gap-2">
   631→        <div className="flex items-end gap-2">
   632→          {keyBtn('E', 'KeyE', 'h-12 w-12 text-sm')}
   633→          {keyBtn('M', 'KeyM', 'h-12 w-12 text-sm')}
   634→          {keyBtn('F', 'KeyF', 'h-12 w-12 text-sm')}
   635→        </div>
   636→        <div className="flex items-end gap-2">
   637→          {keyBtn('⬇', 'KeyC', 'h-12 w-12 text-sm')}
   638→          <button
   639→            className={`hyperion-touchbtn pointer-events-auto flex h-12 w-12 items-center justify-center rounded-full border border-white/25 bg-black/45 text-sm font-black text-white/90 backdrop-blur-[2px] ${sprintOn ? 'text-amber-300' : ''}`}
   640→            data-active={sprintOn}
   641→            onPointerDown={(e) => {
   642→              e.preventDefault();
   643→              setSprintOn(input.toggleVirtualSprint());
   644→            }}
   645→            aria-label="sprint toggle"
   646→          >
   647→            ⚡
   648→          </button>
   649→          {keyBtn('⬆', 'Space', 'h-14 w-14 text-base')}
   650→          {useBtn}
   651→          {digFireBtn}
   652→        </div>
   653→      </div>
   654→    </div>
   655→  );
   656→}
