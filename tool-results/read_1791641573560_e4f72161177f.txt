    10→function ItemIcon({ id, size = 40 }: { id: ItemId; size?: number }) {

    13→  return (
    14→    <img
    15→      src={src}
    16→      alt={ITEMS[id].name}

    20→      className="drop-shadow-[2px_2px_0_rgba(0,0,0,0.45)]"
    21→      draggable={false}
    22→    />

   198→/** Weather chip — clear / overcast / rain with animated droplets + wet-roads tag. */

   200→  const rain = weather.state === 'rain';

   202→  const wet = weather.wet > 0.3;

   204→    <div
   205→      className={`flex items-center gap-2 rounded-lg border px-2.5 py-1 backdrop-blur-[2px] transition-colors duration-700 ${
   206→        rain
   207→          ? 'border-sky-300/30 bg-[#0d1420]/70'
   208→          : over
   209→            ? 'border-white/15 bg-black/55'
   210→            : 'border-amber-300/20 bg-black/45'

   216→          <span className="absolute left-[3px] top-0 hyperion-drop h-[5px] w-[2px] rounded-full bg-sky-300" />
