   306→/** Vehicle speedometer — SVG arc gauge + hull integrity + tank cannon reload. */
   307→function Speedo({ driving }: { driving: NonNullable<HudState['driving']> }) {
   308→  const max = driving.tank ? 60 : 130;
   309→  const pct = Math.max(0, Math.min(1, driving.speed / max));
   310→  const angle = (-90 + pct * 180) * (Math.PI / 180);
   311→  const needleLen = 44;
   312→  const nx = 75 + needleLen * Math.sin(angle);
   313→  const ny = 74 - needleLen * Math.cos(angle);
   314→  const hpct = Math.max(0, Math.min(1, driving.health / (driving.tank ? 400 : 100)));
   315→  const lowHealth = hpct < 0.3;
   316→  return (
   317→    <div
   318→      className={`mt-1 rounded-lg border px-2 pb-1.5 pt-1 backdrop-blur-[2px] ${
   319→        driving.tank ? 'border-lime-400/40 bg-[#0d1206]/75' : 'border-white/15 bg-black/60'
   320→      }`}
   321→    >
   322→      <svg width="150" height="88" viewBox="0 0 150 88">
   323→        {/* gauge arc */}
   324→        <path d="M 23 74 A 52 52 0 0 1 127 74" fill="none" stroke="rgba(255,255,255,0.13)" strokeWidth="7" strokeLinecap="round" />
   325→        <path
   326→          d="M 23 74 A 52 52 0 0 1 127 74"
   327→          fill="none"
   328→          stroke={driving.tank ? '#a3e635' : '#fbbf24'}
   329→          strokeWidth="7"
   330→          strokeLinecap="round"
   331→          strokeDasharray={`${pct * 163.4} 400`}
   332→        />
   333→        {/* ticks */}
   334→        {[0, 0.25, 0.5, 0.75, 1].map((t, i) => {
   335→          const a = (-90 + t * 180) * (Math.PI / 180);
   336→          const x1 = 75 + 40 * Math.sin(a);
   337→          const y1 = 74 - 40 * Math.cos(a);
   338→          const x2 = 75 + 45 * Math.sin(a);
   339→          const y2 = 74 - 45 * Math.cos(a);
   340→          return <line key={i} x1={x1} y1={y1} x2={x2} y2={y2} stroke="rgba(255,255,255,0.4)" strokeWidth="1.6" />;
   341→        })}
   342→        {/* needle */}
   343→        <line x1="75" y1="74" x2={nx} y2={ny} stroke="#ffffff" strokeWidth="2.4" strokeLinecap="round" />
   344→        <circle cx="75" cy="74" r="4" fill="#ffffff" />
   345→        <text x="75" y="58" textAnchor="middle" fill="#ffffff" fontSize="19" fontWeight="800" className="tabular-nums">
   346→          {Math.round(driving.speed)}
   347→        </text>
   348→        <text x="75" y="86" textAnchor="middle" fill="rgba(255,255,255,0.5)" fontSize="8" letterSpacing="3">
   349→          KM/H
   350→        </text>
   351→      </svg>
   352→      {/* hull integrity */}
   353→      <div className="mt-0.5 h-1.5 overflow-hidden rounded-full bg-white/10">
   354→        <div
   355→          className={`h-full rounded-full transition-[width] duration-300 ${
   356→            lowHealth ? 'bg-gradient-to-r from-red-600 to-orange-400' : 'bg-gradient-to-r from-sky-400/80 to-emerald-400'
   357→          }`}
   358→          style={{ width: `${hpct * 100}%` }}
   359→        />
   360→      </div>
   361→      <div className="mt-1 flex items-center justify-between">
   362→        <span className={`text-[9px] font-black uppercase tracking-[0.14em] ${driving.tank ? 'text-lime-300/90' : 'text-amber-300/90'}`}>
   363→          {driving.name}
   364→        </span>
   365→        {driving.tank && (
   366→          <span
   367→            className={`rounded px-1.5 py-0.5 text-[8px] font-black uppercase tracking-widest ${
   368→              driving.reload >= 1 ? 'bg-lime-400/20 text-lime-300' : 'bg-white/10 text-white/45'
   369→            }`}
   370→          >
   371→            {driving.reload >= 1 ? '● LMB · FIRE' : `RELOAD ${Math.round(driving.reload * 100)}%`}
   372→          </span>
   373→        )}
   374→      </div>
   375→      {driving.tank && driving.reload < 1 && (
   376→        <div className="mt-0.5 h-1 overflow-hidden rounded-full bg-white/10">
   377→          <div className="h-full rounded-full bg-lime-400/70 transition-[width] duration-100" style={{ width: `${driving.reload * 100}%` }} />
   378→        </div>
   379→      )}
   380→    </div>
   381→  );
   382→}
