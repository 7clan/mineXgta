    62→  private bodyGeo(w: number, h: number, d: number, r: number): RoundedBoxGeometry {
    63→    const key = `${w}|${h}|${d}|${r}`;
    64→    let g = this.bodyGeoCache.get(key);
    65→    if (!g) {
    66→      g = new RoundedBoxGeometry(w, h, d, 3, r);
    67→      this.bodyGeoCache.set(key, g);
    68→      this.geometries.push(g);
    69→    }
    70→    return g;
    71→  }

    73→  private buildMesh(type: CarType, color: number): { group: THREE.Group; wheels: THREE.Mesh[]; lightbar: THREE.Mesh[] } {
    74→    const group = new THREE.Group();
    75→    const paint = new THREE.MeshStandardMaterial({ color, roughness: 0.28, metalness: 0.65 });
    76→    const dark = new THREE.MeshStandardMaterial({ color: 0x14161a, roughness: 0.55, metalness: 0.3 });
    77→    const glass = new THREE.MeshStandardMaterial({ color: 0x0e1418, roughness: 0.08, metalness: 0.9 });
    78→    const tireMat = new THREE.MeshStandardMaterial({ color: 0x101012, roughness: 0.95 });
    79→    const hubMat = new THREE.MeshStandardMaterial({ color: 0xb8bcc2, roughness: 0.3, metalness: 0.85 });
    80→    const headMat = new THREE.MeshStandardMaterial({ color: 0xfff6d8, emissive: 0xfff0c0, emissiveIntensity: 1.4 });
    81→    const tailMat = new THREE.MeshStandardMaterial({ color: 0x550000, emissive: 0xff2a1a, emissiveIntensity: 1.1 });
    82→
    83→    const dims =
    84→      type === 'sports'
    85→        ? { w: 1.92, h: 0.5, len: 4.4, cabinH: 0.42, cabinLen: 1.9, ride: 0.42, wheel: 0.33 }
    86→        : type === 'suv'
    87→          ? { w: 2.02, h: 0.85, len: 4.9, cabinH: 0.72, cabinLen: 2.6, ride: 0.6, wheel: 0.4 }
    88→          : { w: 1.88, h: 0.62, len: 4.6, cabinH: 0.55, cabinLen: 2.3, ride: 0.52, wheel: 0.35 };
    89→
    90→    const hull = new THREE.Mesh(this.bodyGeo(dims.w, dims.h, dims.len, 0.22), paint);
    91→    hull.position.y = dims.ride;
    92→    hull.castShadow = true;
    93→    group.add(hull);
    94→
    95→    const cabin = new THREE.Mesh(this.bodyGeo(dims.w * 0.88, dims.cabinH, dims.cabinLen, 0.16), glass);
    96→    cabin.position.set(0, dims.ride + dims.h / 2 + dims.cabinH / 2 - 0.06, -0.25);
    97→    cabin.castShadow = true;
    98→    group.add(cabin);
    99→
   100→    const roof = new THREE.Mesh(this.bodyGeo(dims.w * 0.8, 0.06, dims.cabinLen * 0.72, 0.02), paint);
   101→    roof.position.set(0, dims.ride + dims.h / 2 + dims.cabinH - 0.05, -0.25);
   102→    group.add(roof);
   103→
   104→    // bumpers + grille
   105→    const bumperF = new THREE.Mesh(this.bodyGeo(dims.w * 0.96, 0.18, 0.22, 0.06), dark);
   106→    bumperF.position.set(0, dims.ride - dims.h * 0.18, dims.len / 2 - 0.02);
   107→    const bumperR = new THREE.Mesh(this.bodyGeo(dims.w * 0.96, 0.18, 0.22, 0.06), dark);
   108→    bumperR.position.set(0, dims.ride - dims.h * 0.18, -dims.len / 2 + 0.02);
   109→    group.add(bumperF, bumperR);
   110→
   111→    // lights
   112→    for (const s of [-1, 1]) {
   113→      const hl = new THREE.Mesh(new THREE.BoxGeometry(0.3, 0.12, 0.06), headMat);
   114→      hl.position.set(s * dims.w * 0.3, dims.ride + dims.h * 0.18, dims.len / 2 + 0.02);
   115→      const tl = new THREE.Mesh(new THREE.BoxGeometry(0.34, 0.12, 0.06), tailMat);
   116→      tl.position.set(s * dims.w * 0.3, dims.ride + dims.h * 0.2, -dims.len / 2 - 0.02);
   117→      group.add(hl, tl);
   118→    }
   119→
   120→    if (type === 'taxi') {
   121→      const sign = new THREE.Mesh(new THREE.BoxGeometry(0.62, 0.18, 0.3), new THREE.MeshStandardMaterial({ color: 0xffd23f, emissive: 0xffd23f, emissiveIntensity: 0.7 }));
   122→      sign.position.set(0, dims.ride + dims.h / 2 + dims.cabinH + 0.02, -0.2);
   123→      group.add(sign);
   124→    }
   125→
   126→    const lightbar: THREE.Mesh[] = [];
   127→    if (type === 'police') {
   128→      // Black door panels + trunk stripe over white paint
   129→      const stripeMat = new THREE.MeshStandardMaterial({ color: 0x101215, roughness: 0.4, metalness: 0.5 });
   130→      const hood = new THREE.Mesh(this.bodyGeo(dims.w * 0.9, 0.04, 1.1, 0.02), stripeMat);
   131→      hood.position.set(0, dims.ride + dims.h / 2 + 0.03, 1.45);
   132→      const trunk = new THREE.Mesh(this.bodyGeo(dims.w * 0.9, 0.04, 0.9, 0.02), stripeMat);
   133→      trunk.position.set(0, dims.ride + dims.h / 2 + 0.03, -1.7);
   134→      group.add(hood, trunk);
   135→      // Lightbar: red + blue flashing modules
   136→      const red = new THREE.MeshStandardMaterial({ color: 0x5a0806, emissive: 0xff2418, emissiveIntensity: 2 });
   137→      const blue = new THREE.MeshStandardMaterial({ color: 0x061a5a, emissive: 0x2a5aff, emissiveIntensity: 0.2 });
   138→      const r = new THREE.Mesh(new THREE.BoxGeometry(0.5, 0.13, 0.28), red);
   139→      const b = new THREE.Mesh(new THREE.BoxGeometry(0.5, 0.13, 0.28), blue);
   140→      const barY = dims.ride + dims.h / 2 + dims.cabinH + 0.04;
   141→      r.position.set(-0.3, barY, -0.2);
   142→      b.position.set(0.3, barY, -0.2);
   143→      group.add(r, b);
   144→      lightbar.push(r, b);
   145→    }
   146→
   147→    const wheels: THREE.Mesh[] = [];
   148→    const tireGeo = new THREE.CylinderGeometry(dims.wheel, dims.wheel, 0.26, 14);
   149→    tireGeo.rotateZ(Math.PI / 2);
   150→    const hubGeo = new THREE.CylinderGeometry(dims.wheel * 0.55, dims.wheel * 0.55, 0.27, 10);
   151→    hubGeo.rotateZ(Math.PI / 2);
   152→    this.geometries.push(tireGeo, hubGeo);
   153→    for (const [sx, sz] of [
   154→      [-1, 1],
   155→      [1, 1],
   156→      [-1, -1],
   157→      [1, -1],
   158→    ]) {
   159→      const wm = new THREE.Mesh(tireGeo, tireMat);
   160→      const hub = new THREE.Mesh(hubGeo, hubMat);
   161→      wm.add(hub);
   162→      wm.position.set(sx * (dims.w / 2 - 0.02), dims.wheel, sz * (dims.len / 2 - 0.85));
   163→      wm.castShadow = true;
   164→      group.add(wm);
   165→      wheels.push(wm);
   166→    }
   167→    return { group, wheels, lightbar };
   168→  }

   183→        car.yaw = this.rng() < 0.5 ? 0 : Math.PI;
   184→        car.yawVis = car.yaw;
   185→        car.group.rotation.y = car.yaw;
   186→        placed++;
   187→      }
   188→    }
   189→    // A couple of cars parallel-parked along roadsides.
   190→    for (let i = 0; i < 4; i++) {
   191→      const road = (Math.floor(this.rng() * (GRID_HALF * 2 + 1)) - GRID_HALF) * BLOCK;
   192→      const car = this.makeCar(this.pickType(), 'parked');
   193→      car.group.position.set(road + ROAD_W / 2 - 1.4, 0, roadsideZ * (i + 1) * 0.35 % 300 - 150);
   194→      car.yaw = 0;
   195→      car.yawVis = 0;
   196→      car.group.rotation.y = 0;
   197→    }

   224→      yawVis: 0,
   225→      health: 100,
   226→      burnT: 0,
   227→      honkT: 0,
   228→      blockedT: 0,

   251→  private spawnCar(_mode: 'traffic'): void {
   252→    const car = this.makeCar(this.pickType(), 'traffic');
   253→    // Random road + lane + direction (right-hand traffic)
   254→    const vertical = this.rng() < 0.5;
   255→    const road = (Math.floor(this.rng() * (GRID_HALF * 2 + 1)) - GRID_HALF) * BLOCK;
   256→    const dirRoll = Math.floor(this.rng() * 4) as 0 | 1 | 2 | 3;
   257→    car.dir = dirRoll;
   258→    if (vertical) {
   259→      car.road = road;
   260→      car.lane = road + (dirRoll === 0 ? -LANE_OFF : LANE_OFF);
   261→      car.group.position.set(car.lane, 0, (this.rng() - 0.5) * 360);
   262→      car.yaw = dirRoll === 0 ? 0 : Math.PI;
   263→    } else {
   264→      car.road = road;
   265→      car.lane = road + (dirRoll === 1 ? LANE_OFF : -LANE_OFF);
   266→      car.group.position.set((this.rng() - 0.5) * 360, 0, car.lane);
   267→      car.yaw = dirRoll === 1 ? Math.PI / 2 : -Math.PI / 2;
   268→    }
   269→    car.yawVis = car.yaw;
   270→    car.group.rotation.y = car.yaw;
   271→  }

   273→  /** Spawn a police cruiser on a road ~90-150m from the player (pursuit AI). */
   274→  spawnPolice(playerPos: THREE.Vector3): Car | null {
   275→    for (let tries = 0; tries < 14; tries++) {
   276→      const vertical = this.rng() < 0.5;
   277→      const road = (Math.floor(this.rng() * (GRID_HALF * 2 + 1)) - GRID_HALF) * BLOCK;
   278→      const dirSign = this.rng() < 0.5 ? 1 : -1;
   279→      let x: number;
   280→      let z: number;
   281→      if (vertical) {
   282→        x = road + (this.rng() < 0.5 ? LANE_OFF : -LANE_OFF);
   283→        z = playerPos.z + dirSign * (90 + this.rng() * 60);
   284→      } else {
   285→        z = road + (this.rng() < 0.5 ? LANE_OFF : -LANE_OFF);
   286→        x = playerPos.x + dirSign * (90 + this.rng() * 60);
   287→      }
   288→      if (Math.abs(x) > GRID_HALF * BLOCK - 8 || Math.abs(z) > GRID_HALF * BLOCK - 8) continue;
   289→      const car = this.makeCar('police', 'police');
   290→      car.group.position.set(x, 0, z);
   291→      const toP = playerPos.clone().sub(car.group.position);
   292→      car.yaw = Math.atan2(toP.x, toP.z);
   293→      car.yawVis = car.yaw;
   294→      car.group.rotation.y = car.yaw;
   295→      car.targetSpeed = 27;
   296→      return car;
   297→    }
   298→    return null;
   299→  }

   301→  /** Remove the farthest pursuit units beyond `keep`. */
   302→  despawnExtraPolice(keep: number): void {
   303→    const units = this.cars.filter((c) => c.state === 'police');
   304→    if (units.length <= keep) return;
   305→    units.sort((a, b) => b.group.position.distanceTo(this.playerRef) - a.group.position.distanceTo(this.playerRef));
   306→    for (let i = 0; i < units.length - keep; i++) {
   307→      const c = units[i];
   308→      this.scene.remove(c.group);
   309→      const idx = this.cars.indexOf(c);
   310→      if (idx >= 0) this.cars.splice(idx, 1);
   311→    }
   312→  }

   314→  /** Police pursuit: seek the player, shove traffic aside, ram, unstick by reversing. */
   315→  private drivePolice(car: Car, dt: number, world: World, target: THREE.Vector3): void {
   316→    const p = car.group.position;
   317→    // Flash the lightbar
   318→    car.flashT += dt;
   319→    const phase = Math.floor(car.flashT * 7) % 2;
   320→    if (car.lightbar.length === 2) {
   321→      (car.lightbar[0].material as THREE.MeshStandardMaterial).emissiveIntensity = phase === 0 ? 3.4 : 0.12;
   322→      (car.lightbar[1].material as THREE.MeshStandardMaterial).emissiveIntensity = phase === 1 ? 3.4 : 0.12;
   323→    }
   324→
   325→    const toT = target.clone().sub(p);
   326→    toT.y = 0;
   327→    const dist = toT.length();
   328→
   329→    if (car.reverseT > 0) {
   330→      car.reverseT -= dt;
   331→      car.speed = Math.max(car.speed - 18 * dt, -7);
   332→      car.yaw -= 1.25 * dt;
   333→      const heading = new THREE.Vector3(Math.sin(car.yaw), 0, Math.cos(car.yaw));
   334→      p.addScaledVector(heading, car.speed * dt);
   335→      car.group.rotation.y = car.yaw;
   336→      this.spinWheels(car, dt, car.speed);
   337→      return;
   338→    }
   339→
   340→    // Steer toward the player
   341→    const desiredYaw = Math.atan2(toT.x, toT.z);
   342→    let dy = desiredYaw - car.yaw;
   343→    while (dy > Math.PI) dy -= Math.PI * 2;
   344→    while (dy < -Math.PI) dy += Math.PI * 2;
   345→    car.yaw += dy * Math.min(1, (3.4 + Math.abs(car.speed) * 0.05) * dt);
   346→
   347→    const wantSpeed = dist > 14 ? 30 : Math.max(5, dist * 2.1);
   348→    car.speed += (wantSpeed - car.speed) * Math.min(1, 1.7 * dt);
   349→
   350→    // Move with building collision probe
   351→    const heading = new THREE.Vector3(Math.sin(car.yaw), 0, Math.cos(car.yaw));
   352→    const next = p.clone().addScaledVector(heading, car.speed * dt);
   353→    const probe = next.clone();
   354→    probe.y = 1;
   355→    const before = probe.clone();
   356→    world.resolveCircleXZ(probe, 1.35, 0.2, 1.6);
   357→    if (probe.distanceToSquared(before) > 1e-6) {
   358→      p.x += (probe.x - before.x) * 0.55;
   359→      p.z += (probe.z - before.z) * 0.55;
   360→      car.stuckT += dt;
   361→      if (car.stuckT > 1.1) {
   362→        car.stuckT = 0;
   363→        car.reverseT = 0.75;
   364→      }
   365→      car.speed *= 0.85;
   366→    } else {
   367→      car.stuckT = Math.max(0, car.stuckT - dt);
   368→      p.copy(next);
   369→    }
   370→
   371→    // Shove civilian traffic aside (GTA-style PIT maneuver energy)
   372→    for (const o of this.cars) {
   373→      if (o === car || o.state === 'wrecked' || o.state === 'falling' || o.state === 'police') continue;
   374→      const d = o.group.position.distanceTo(p);
   375→      if (d < 3.2 && d > 0.01) {
   376→        const push = o.group.position.clone().sub(p).setY(0).normalize().multiplyScalar((3.2 - d) * 0.55);
   377→        o.group.position.add(push);
   378→        if (o.state === 'traffic') o.state = 'parked';
   379→        car.speed *= 0.94;
   380→      }
   381→    }
   382→
   383→    // Ram the player on foot
   384→    if (dist < 2.7 && Math.abs(car.speed) > 5) {
   385→      world.damagePlayer(7);
   386→      world.shake(0.4);
   387→      world.audio.thud(0.05);
   388→      car.speed *= 0.45;
   389→    }
   390→
   391→    car.group.rotation.y = car.yaw;
   392→    this.spinWheels(car, dt, car.speed);
   393→  }

   395→  nearest(pos: THREE.Vector3, maxDist: number): Car | null {
   396→    let best: Car | null = null;
   397→    let bd = maxDist * maxDist;
   398→    for (const c of this.cars) {
   399→      if (c.state === 'wrecked' || c.state === 'falling') continue;
   400→      const d = c.group.position.distanceToSquared(pos);
   401→      if (d < bd) {
   402→        bd = d;
   403→        best = c;
   404→      }
   405→    }
   406→    return best;
   407→  }

   409→  ignite(car: Car): void {
   410→    if (car.state === 'burning' || car.state === 'wrecked') return;
   411→    if (car.state === 'player') return;
   412→    car.state = 'burning';
   413→    car.burnT = 3.4 + this.rng() * 1.8;
   414→  }

   416→  /** Rain dousing: a burning car caught in heavy rain goes out (still damaged). */
   417→  extinguish(car: Car): void {
   418→    if (car.state !== 'burning') return;
   419→    car.state = car.type === 'police' ? 'police' : 'parked';
   420→    car.burnT = 0;
   421→    car.health = Math.max(14, car.health);
   422→    car.targetSpeed = car.type === 'police' ? 27 : car.targetSpeed;
   423→    // Steam puff (state is no longer 'burning', so game renders smoke only)
   424→    this.onSmoke?.(car);
   425→  }
   426→
   427→  explode(car: Car): void {
   428→    car.state = 'wrecked';
   429→    car.speed = 0;
   430→    const p = car.group.position;
   431→    this.charCar(car);
   432→    this.onExplosion?.(new THREE.Vector3(p.x, p.y + 0.8, p.z), 5.0);
   433→    // Tilt the wreck
   434→    car.group.rotation.z = (this.rng() - 0.5) * 0.25;
   435→    car.group.rotation.x = (this.rng() - 0.5) * 0.18;
   436→  }

   438→  private charCar(car: Car): void {
   439→    car.group.traverse((o) => {
   440→      if (o instanceof THREE.Mesh) {
   441→        const m = o.material as THREE.MeshStandardMaterial;
   442→        if (m && m.color) {
   443→          m.color.multiplyScalar(0.18);
   444→          m.roughness = 0.95;
   445→          m.metalness = 0.1;
   446→          if (m.emissive) m.emissive.setHex(0x000000);
   447→        }
   448→      }
   449→    });
   450→  }

   457→    car.state = 'parked';
   458→    car.speed = 0;
   459→    const side = new THREE.Vector3(Math.cos(car.yaw), 0, -Math.sin(car.yaw));
   460→    playerPos.copy(car.group.position).addScaledVector(side, 2.1);

   464→  /** Drive as player: throttle/brake/steer arcade physics. */
   465→  drivePlayer(car: Car, dt: number, throttle: number, steer: number, handbrake: boolean, world: World): void {
   466→    const accel = car.type === 'sports' ? 11 : car.type === 'suv' ? 7.5 : 8.5;
   467→    const maxSpeed = car.type === 'sports' ? 34 : car.type === 'suv' ? 24 : 27;
   468→    if (throttle > 0) car.speed += accel * throttle * dt;
   469→    else if (throttle < 0) car.speed -= (car.speed > 0 ? 26 : 10) * dt;
   470→    car.speed = Math.max(-9, Math.min(maxSpeed, car.speed));
   471→    const drag = handbrake ? 3.4 : 0.55;
   472→    car.speed -= car.speed * drag * dt * Math.abs(car.speed) * 0.012;
   473→    if (Math.abs(throttle) < 0.05) car.speed -= Math.sign(car.speed) * 4.2 * dt;
   474→
   475→    const speedFactor = Math.min(1, Math.abs(car.speed) / 14);
   476→    car.yaw += steer * 1.9 * speedFactor * dt * Math.sign(car.speed || 1) * (handbrake ? 1.5 : 1);
   477→    const heading = new THREE.Vector3(Math.sin(car.yaw), 0, Math.cos(car.yaw));
   478→    const next = car.group.position.clone().addScaledVector(heading, car.speed * dt);
   479→
   480→    // Collisions with buildings (circle vs AABB).
   481→    const r = 1.35;
   482→    const probe = next.clone();
   483→    probe.y = 1;
   484→    const before = probe.clone();
   485→    world.resolveCircleXZ(probe, r, 0.2, 1.6);
   486→    if (probe.distanceToSquared(before) > 1e-6) {
   487→      // Hit something solid
   488→      if (Math.abs(car.speed) > 9) {
   489→        car.health -= Math.abs(car.speed) * 1.1;
   490→        world.audio.thud(0.1);
   491→        world.spawnDebris(probe, 4, new THREE.Color(0x888888), 0.7);
   492→        world.shake(Math.min(0.5, Math.abs(car.speed) * 0.02));
   493→      }
   494→      car.speed *= -0.18;
   495→    } else {
   496→      car.group.position.copy(next);
   497→    }
   498→
   499→    // Car-vs-car
   500→    for (const o of this.cars) {
   501→      if (o === car) continue;
   502→      const d = o.group.position.distanceTo(car.group.position);
   503→      if (d < 3.4 && d > 0.01) {
   504→        const push = car.group.position.clone().sub(o.group.position).normalize().multiplyScalar((3.4 - d) * 0.5);
   505→        car.group.position.add(push);
   506→        o.group.position.sub(push);
   507→        if (Math.abs(car.speed) > 10) {
   508→          o.health -= 24;
   509→          car.health -= 12;
   510→          world.audio.thud(0.2);
   511→          if (o.state === 'traffic') o.state = 'parked';
   512→          if (o.health <= 0 && o.state !== 'burning' && o.state !== 'wrecked') this.ignite(o);
   513→        }
   514→        car.speed *= 0.72;
   515→      }
   516→    }
   517→
   518→    // Run over NPCs
   519→    if (Math.abs(car.speed) > 5) {
   520→      this.onRagdollNPCs?.(car.group.position, 1.9);
   521→    }
   522→
   523→    car.group.rotation.y = car.yaw;
   524→    this.spinWheels(car, dt, car.speed);
   525→  }

   527→  update(car0: Car | null, dt: number, world: World, playerPos: THREE.Vector3): void {
   528→    const burning: Car[] = [];
   529→    for (const car of this.cars) {
   530→      const p = car.group.position;
   531→
   532→      // Falling into dug road holes
   533→      if (car.state !== 'falling' && car.state !== 'wrecked' && world.floorHeightAt(p.x, p.z, p.y + 0.4) < -1.5) {
   534→        if (car.state === 'player') continue;
   535→        car.state = 'falling';
   536→        car.fallVel = 0;
   537→      }
   538→      if (car.state === 'falling') {
   539→        car.fallVel -= 22 * dt;
   540→        p.y += car.fallVel * dt;
   541→        if (p.y < -2.4) {
   542→          this.explode(car);
   543→        }
   544→        continue;
   545→      }
   546→
   547→      if (car.state === 'burning') {
   548→        burning.push(car);
   549→        car.burnT -= dt;
   550→        car.smokeAcc += dt;
   551→        if (car.health > 25) car.health = 25;
   552→        if (car.burnT <= 0) this.explode(car);
   553→      }
   554→      if (car.health < 30 && car.state !== 'wrecked' && car.state !== 'burning') {
   555→        car.smokeAcc += dt;
   556→      }
   557→
   558→      if (car.state === 'traffic' && car !== car0) {
   559→        this.driveTraffic(car, dt, world, playerPos);
   560→      }
   561→      if (car.state === 'police' && car !== car0) {
   562→        this.drivePolice(car, dt, world, playerPos);
   563→      }
   564→      if (car.state !== 'wrecked') {
   565→        p.y = Math.max(p.y, 0);
   566→      }
   567→      // wheel spin for parked/burning (traffic handled in driveTraffic)
   568→      if (car.state === 'parked' || car.state === 'burning') this.spinWheels(car, dt, 0);
   569→    }
   570→
   571→    // Smoke for damaged/burning cars handled by game via callback below.
   572→    for (const car of this.cars) {
   573→      if (car.smokeAcc > 0.09 && (car.state === 'burning' || car.health < 30) && car.state !== 'wrecked') {
   574→        car.smokeAcc = 0;
   575→        this.onSmoke?.(car);
   576→      }
   577→    }
   578→    void burning;
   579→  }

   583→  private driveTraffic(car: Car, dt: number, world: World, playerPos: THREE.Vector3): void {
   584→    const [dx, dz] = DIR_VEC[car.dir];
   585→    const p = car.group.position;
   586→
   587→    // Ahead checks: other cars + player
   588→    let brake = false;
   589→    const ahead = new THREE.Vector3(p.x + dx * 7, 0, p.z + dz * 7);
   590→    for (const o of this.cars) {
   591→      if (o === car) continue;
   592→      const rel = o.group.position.clone().sub(p);
   593→      const along = rel.x * dx + rel.z * dz;
   594→      const lateral = Math.abs(rel.x * dz - rel.z * dx);
   595→      if (along > 0 && along < 8.5 && lateral < 1.6 && o.state !== 'wrecked') {
   596→        brake = true;
   597→        break;
   598→      }
   599→    }
   600→    const prel = playerPos.clone().sub(p);
   601→    const pAlong = prel.x * dx + prel.z * dz;
   602→    const pLat = Math.abs(prel.x * dz - prel.z * dx);
   603→    if (pAlong > 0 && pAlong < 6.5 && pLat < 1.8 && !brake) brake = true;
   604→
   605→    if (brake) {
   606→      car.blockedT += dt;
   607→      car.speed = Math.max(0, car.speed - 16 * dt);
   608→      if (car.blockedT > 2.2 && car.honkT <= 0) {
   609→        world.audio.horn(this.far(p));
   610→        car.honkT = 3.5 + this.rng() * 3;
   611→      }
   612→    } else {
   613→      car.blockedT = 0;
   614→      car.speed += (car.targetSpeed - car.speed) * Math.min(1, 1.4 * dt);
   615→    }
   616→    car.honkT -= dt;
   617→
   618→    p.x += dx * car.speed * dt;
   619→    p.z += dz * car.speed * dt;
   620→
   621→    // Lane keeping
   622→    if (car.dir === 0 || car.dir === 2) p.x += (car.lane - p.x) * Math.min(1, 3 * dt);
   623→    else p.z += (car.lane - p.z) * Math.min(1, 3 * dt);
   624→
   625→    // Intersections: random turns
   626→    const nearestCross = Math.round(p.x / BLOCK) * BLOCK;
   627→    const nearestCrossZ = Math.round(p.z / BLOCK) * BLOCK;
   628→    const onVertical = car.dir === 0 || car.dir === 2;
   629→    const crossCoord = onVertical ? nearestCrossZ : nearestCross;
   630→    const alongPos = onVertical ? p.z : p.x;
   631→    const passed = Math.abs(alongPos - crossCoord) < 0.5 + car.speed * dt;
   632→    if (passed && Math.abs(crossCoord) <= GRID_HALF * BLOCK) {
   633→      const turnKey = `${car.id}:${Math.round(crossCoord)}`;
   634→      if (car.honkT !== -999 && (car as unknown as { lastTurn?: string }).lastTurn !== turnKey) {
   635→        (car as unknown as { lastTurn?: string }).lastTurn = turnKey;
   636→        const roll = this.rng();
   637→        if (roll < 0.45) {
   638→          // turn
   639→          const left = this.rng() < 0.5;
   640→          if (onVertical) {
   641→            car.dir = left ? 1 : 3;
   642→            car.road = crossCoord;
   643→            car.lane = crossCoord + (car.dir === 1 ? LANE_OFF : -LANE_OFF);
   644→            p.z = crossCoord + dz * 0.4;
   645→          } else {
   646→            car.dir = left ? 2 : 0;
   647→            car.road = crossCoord;
   648→            car.lane = crossCoord + (car.dir === 0 ? -LANE_OFF : LANE_OFF);
   649→            p.x = crossCoord + dx * 0.4;
   650→          }
   651→        }
   652→      }
   653→    }
   654→
   655→    // Out of city bounds: turn around
   656→    if (Math.abs(p.x) > GRID_HALF * BLOCK + 6 || Math.abs(p.z) > GRID_HALF * BLOCK + 6) {
   657→      car.dir = ((car.dir + 2) % 4) as 0 | 1 | 2 | 3;
   658→      this.relane(car);
   659→    }
   660→
   661→    const targetYaw = Math.atan2(dx, dz);
   662→    let dy = targetYaw - car.yaw;
   663→    while (dy > Math.PI) dy -= Math.PI * 2;
   664→    while (dy < -Math.PI) dy += Math.PI * 2;
   665→    car.yaw += dy * Math.min(1, 8 * dt);
   666→    car.yawVis = car.yaw;
   667→    car.group.rotation.y = car.yaw;
   668→    this.spinWheels(car, dt, car.speed);
   669→  }
   670→
   671→  private relane(car: Car): void {
   672→    const [dx, dz] = DIR_VEC[car.dir];
   673→    if (dz !== 0) car.lane = car.road + (dz > 0 ? -LANE_OFF : LANE_OFF);
   674→    else car.lane = car.road + (dx > 0 ? LANE_OFF : -LANE_OFF);
   675→  }

   677→  private spinWheels(car: Car, dt: number, speed: number): void {
   678→    for (const w of car.wheels) {
   679→      w.rotation.x -= (speed * dt) / 0.35;
   680→    }
   681→  }
